function doGet(e) {
  // パラメータ取得
  const schoolName = (e.parameter.schoolName || "").toString().trim();
  const kakomonName = (e.parameter.kakomonName || "").toString().trim();
  const inputPasscode = (e.parameter.passcode || "").toString().trim();

  // 全体管理ブックのID（Adminシートが存在するスプレッドシートID）
  const MASTER_ADMIN_ID = e.parameter.id || "1tm6YBQ52wCqMjM4HSrpo1pp1gwNd9YpF-liLb-Cvqec";
  const ADMIN_SHEET_NAME = e.parameter.adminSheetName || "admnin";

  try {
    // 1. 管理シート（Admin）を読み込む
    const masterSs = SpreadsheetApp.openById(MASTER_ADMIN_ID);
    let adminSheet = masterSs.getSheetByName(ADMIN_SHEET_NAME) || masterSs.getSheets()[0];
    const adminData = readSheetData(adminSheet);

    // 2. 該当する行を検索（School_nameとKakomon_nameでマッチング）
    const row = adminData.find(d => {
      const s = (d.School_name || d.school_name || "").toString().trim();
      const k = (d.Kakomon_name || d.kakomon_name || "").toString().trim();

      // School_name が指定されていない（空欄）場合は Kakomon_name のみで比較
      if (!schoolName) {
        return k === kakomonName;
      }
      return s === schoolName && k === kakomonName;
    });

    if (!row) {
      return createResponse({ error: `管理シート内に「${kakomonName}」に一致する設定が見つかりません。` });
    }

    // 3. パスコード認証
    const requiredPasscode = (row.Passcode || row.passcode || "").toString().trim();
    if (requiredPasscode !== "") {
      if (inputPasscode === "") {
        return createResponse({ error: "PASSCODE_REQUIRED", message: "パスコードを入力してください。" });
      }
      if (inputPasscode !== requiredPasscode) {
        return createResponse({ error: "PASSCODE_INCORRECT", message: "パスコードが正しくありません。" });
      }
    }

    // 4. 問題ブックのIDと各シート名を取得
    const targetBookId = (row.Kakomon_ID || row.kakomon_id || MASTER_ADMIN_ID).toString().trim();
    const sheetName = row.Kakomon_sheet || row.kakomon_sheet;
    const categorySheetName = row.Category_sheet || row.category_sheet;

    const targetSs = SpreadsheetApp.openById(targetBookId);
    const response = {};

    // 5. 問題シート読み込み
    const qSheet = targetSs.getSheetByName(sheetName);
    if (qSheet) {
      response.quizData = readSheetData(qSheet);
    } else {
      return createResponse({ error: `指定された問題シート「${sheetName}」が存在しません。` });
    }

    // 6. 出題設定（Category_sheet / Config）読み込み
    if (categorySheetName) {
      const cSheet = targetSs.getSheetByName(categorySheetName);
      if (cSheet) {
        const configRaw = readSheetData(cSheet);
        const configMap = {};
        configRaw.forEach(r => {
          const k = r.category || r.Category;
          const v = r.count || r.Count;
          if (k && v) configMap[k] = parseInt(v, 10);
        });
        response.quizConfig = configMap;
      }
    }

    return createResponse(response);

  } catch (err) {
    return createResponse({ error: err.toString() });
  }
}

// スプレッドシートを1行目ヘッダーとしてオブジェクト配列に変換する関数
function readSheetData(sheet) {
  const values = sheet.getDataRange().getValues();
  if (values.length < 2) return [];
  const headers = values[0].map(h => h.toString().trim());
  const data = [];
  for (let i = 1; i < values.length; i++) {
    const row = {};
    for (let j = 0; j < headers.length; j++) {
      row[headers[j]] = values[i][j];
    }
    data.push(row);
  }
  return data;
}

// JSONレスポンス作成関数
function createResponse(data) {
  return ContentService.createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}
